  function starTier(value){
    if(value>=1000) return {count:3,color:'red'};
    if(value>=500)  return {count:2,color:'red'};
    if(value>=100)  return {count:1,color:'red'};
    if(value>=50)   return {count:3,color:'yellow'};
    if(value>=30)   return {count:2,color:'yellow'};
    return {count:1,color:'yellow'};
  }

  function starMarkup(value){
    const tier=starTier(value);
    const glyph=tier.color==='red' ? '★' : '★';
    const color=tier.color==='red' ? '#ff3b30' : '#ffd42a';

    return Array.from({length:tier.count},()=>(
      `<span class="star-glyph" style="color:${color}">${glyph}</span>`
    )).join('');
  }
