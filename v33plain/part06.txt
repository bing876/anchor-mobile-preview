s.w/2-center.x;
    camY=subjectPos.y+subjectPos.h/2-center.y;

    render();
  }

  function settlePinch(commit){
    if(!pinch)return;

    const state=pinch;
    pinch=null;

    const target=commit?1:0;
    const start=state.progress;
    const duration=commit?480:380;
    const begun=performance.now();

    const startVisual=state.fromMode==='two' ? start : 1-start;
    const targetVisual=commit
      ? (state.toMode==='borderless' ? 1 : 0)
      : (state.fromMode==='borderless' ? 1 : 0);

    stopSettle();

    function tick(now){
      const t=Math.min(1,(now-begun)/duration);
      const e=1-Math.pow(1-t,4);
      const p=lerp(start,target,e);
      morphVisualAmount=lerp(startVisual,targetVisual,e);

      layout=blendedLayout(state.fromLayout,state.toLayout,p);

      const subjectPos=layout[state.subject];
      camX=subjectPos.x+subjectPos.w/2-state.anchor.x;
      camY=subjectPos.y+subjectPos.h/2-state.anchor.y;

      render();

      if(t<1){
        settleRAF=requestAnimationFrame(tick);
        return;
      }

      settleRAF=0;
      mode=commit?state.toMode:state.fromMode;
      layout=mode==='two'?twoLayout():borderlessLayout(state.subject);

      surface.classList.toggle('two',mode==='two');
      surface.classList.toggle('borderless',mode==='borderless');
      surface.classList.remove('dragging','pinch-live');

      morphVisualAmount=null;
      clampCamera();
      render();

      if(mode==='borderless'){
        scheduleBorderlessCenter(320);
      }else{
        resetCenterProgress(true);
        scheduleTwoAlignment(140,nearestTwoTargetX(camX));
      }
    }

    settleRAF=requestAnimationFrame(tick);
  }

  function stopSettle(){
    if(settleRAF){
      cancelAnimationFrame(settleRAF);
      settleRAF=0;
    }
  }

  /* =================================================================
     05 · SINGLE-FINGER / MOUSE SCROLLING
     Borderless interaction follows the uploaded v45 behavior:
     vertical gestures suppress tiny x drift; clear side intent releases freely.
     ================================================================= */
  function stopMotion(){
    if(inertiaRAF){cancelAnimationFrame(inertiaRAF);inertiaRAF=0}
    if(snapRAF){cancelAnimationFrame(snapRAF);snapRAF=0}
    if(twoAlignTimer){clearTimeout(twoAlignTimer);twoAlignTimer=0}
    cancelBorderlessCenter(false);
    stopSettle();
  }

  function nearestTwoTargetX(sourceX=camX){
    const pageW=surface.clientWidth;
    const b=cameraBounds();

    const x=Math.max(b.minX,Math.min(b.maxX,sourceX));
    const pageFloat=x/pageW;
    const leftPage=Math.floor(pageFloat);
    const rightPage=Math.ceil(pageFloat);

    const leftX=Math.max(b.minX,Math.min(b.maxX,leftPage*pageW));
    const rightX=Math.max(b.minX,Math.min(b.maxX,rightPage*pageW));

    const leftDistance=Math.abs(x-leftX);
    const rightDistance=Math.abs(rightX-x);

    // Strictly shortest path. No velocity bias.
    return rightDistance < leftDistance ? rightX : leftX;
  }

  function softSnapTwoX(targetX=twoRecoveryTargetX){
    if(mode!=='two') return;

    if(snapRAF){
      cancelAnimationFrame(snapRAF);
      snapRAF=0;
    }

    const b=cameraBounds();
    const fromX=camX;

    const resolvedTarget=targetX===null
      ? nearestTwoTargetX(fromX)
      : Math.max(b.minX,Math.min(b.maxX,targetX));

    twoRecoveryTargetX=resolvedTarget;

    if(Math.abs(resolvedTarget-fromX)<.2){
      camX=resolvedTarget;
      render();
      twoRecoveryTargetX=null;
      return;
    }

    const duration=2400;
    const begun=performance.now();

    function easeInOutCubic(t){
      return t<.5 ? 4*t*t*t : 1-Math.pow(-2*t+2,3)/2;
    }

    function tick(now){
      const t=Math.min(1,(now-begun)/duration);
      const e=easeInOutCubic(t);

      // Only recover horizontal alignment.
      // camY remains fully controlled by normal vertical inertia/scroll.
      camX=lerp(fromX,resolvedTarget,e);
      render();

      if(t<1){
        snapRAF=requestAnimationFrame(tick);
      }else{
        snapRAF=0;
        camX=resolvedTarget;
        twoRecoveryTargetX=null;
        render();
      }
    }

    snapRAF=requestAnimationFrame(tick);
  }

  function scheduleTwoAlignment(delay=320,targetX=twoRecoveryTargetX){
    if(mode!=='two') return;
    if(twoAlignTimer) clearTimeout(twoAlignTimer);

    if(targetX!==null){
      twoRecoveryTargetX=targetX;
    }

    twoAlignTimer=setTimeout(()=>{
      twoAlignTimer=0;
      if(mode==='two' && !dragId && !pinch){
        softSnapTwoX(twoRecoveryTargetX);
      }
    },delay);
  }

  function inertia(ix,iy,settleTwo=false){
    let prev=performance.now();

    function tick(now){
      const dt=Math.min(32,now-prev)/16.67;
      prev=now;

      camX+=ix*dt;
      camY+=iy*dt;

      const {ox,oy}=overshoot();

      if(ox!==0){
        ix+=(-ox*.032)*dt;
        ix*=Math.pow(.84,dt);
      }
      if(oy!==0){
        iy+=(-oy*.032)*dt;
        iy*=Math.pow(.84,dt);
      }

      // A touch less friction than before so two-column gliding
      // feels closer to borderless before slowly recovering.
      const friction=Math.pow(.938,dt);
      ix*=friction;
      iy*=friction;

      render();

      const speed=Math.hypot(ix,iy);
      const edgeDistance=Math.hypot(ox,oy);

      if(speed>.14 || edgeDistance>.40){
        inertiaRAF=requestAnimationFrame(tick);
      }else{
        inertiaRAF=0;
        if(mode==='two'){
          if(!snapRAF){
            twoRecoveryTargetX=null;
            scheduleTwoAlignment(140,nearestTwoTargetX(camX));
          }
        }else if(mode==='borderless'){
          scheduleBorderlessCenter(160);
        }else{
          springToBounds();
        }
      }
    }

    inertiaRAF=requestAnimationFrame(tick);
  }

  surface.addEventListener('pointerdown',e=>{
    if(e.button!==undefined && e.button!==0)return;

    if(!detailOpen){
      const card=e.target.closest?.('.card');
      tapCandidateIndex=card ? Number(card.dataset.index) : -1;
      tapStartX=e.clientX;
      tapStartY=e.clientY;
      tapStartTime=performance.now();
      tapMoved=false;
    }

    pointers.set(e.pointerId,{x:e.clientX,y:e.clientY});
    try{surface.setPointerCapture(e.pointerId)}catch(_){}

    if(pointers.size===1){
      stopMotion();
      twoRecoveryTargetX=null;
      dragId=e.pointerId;
      dragStartX=lastX=e.clientX;
      dragStartY=lastY=e.clientY;
      dragStartCamX=camX;
      dragStartCamY=camY;
      lastT=performance.now();
      velX=velY=0;
      gestureAxis='pending';
      sideIntentFrames=0;
      twoHorizontalUnlocked=false;
      twoVerticalLockedX=null;
      surface.classList.add('dragging');
    }else if(pointers.size===2){
      tapCandidateIndex=-1;
      tapMoved=true;
      dragId=null;
      beginPinch();
    }
  });

  surface.addEventListener('pointermove',e=>{
    if(!pointers.has(e.pointerId))return;
    pointers.set(e.pointerId,{x:e.clientX,y:e.clientY});

    if(pointers.size===2){
      updatePinch();
      e.preventDefault();
      return;
    }

    if(e.pointerId!==dragId || pinch)return;

    let dx=e.clientX-dragStartX;
    let dy=e.clientY-dragStartY;
    const stepX=e.clientX-lastX;
    const stepY=e.clientY-lastY;

    if(!tapMoved && Math.hypot(e.clientX-tapStartX,e.clientY-tapStartY)>9){
      tapMoved=true;
    }

    if(mode==='borderless' && Math.hypot(dx,dy)>=4){
      cancelBorderlessCenter(true);
    }

    /* -------------------------------------------------------------
       BORDERLESS
       Keep the original direct finger-following behavior.
       ------------------------------------------------------------- */
    if(mode==='borderless'){
      applyDragResistance(
        dragStartCamX-dx,
        dragStartCamY-dy
      );

      const now=performance.now();
      const dt=Math.max(1,now-lastT);
      velX=-(e.clientX-lastX)/dt;
      velY=-(e.clientY-lastY)/dt;

      lastX=e.clientX;
      lastY=e.clientY;
      lastT=now;

      clampCamera();
      render();
      e.preventDefault();
      return;
    }

    const absDX=Math.abs(dx);
    const absDY=Math.abs(dy);
    const absStepX=Math.abs(stepX);
    const absStepY=Math.abs(stepY);

    /* -------------------------------------------------------------
       TWO-COLUMN INTENT
       Default vertical scroll is horizontally fixed.
       Horizontal intent unlocks the same direct canvas movement used
       by borderless mode.
       ------------------------------------------------------------- */
    if(mode==='two'){
      if(gestureAxis==='pending'){
        const travel=Math.hypot(dx,dy);

        if(travel>=4){
          if(absDY>absDX*1.15){
            gestureAxis='vertical';
            twoVerticalLockedX=nearestTwoTargetX(camX);
          }else if(absDX>absDY*1.12 && absDX>=6){
            gestureAxis='free';
            twoHorizontalUnlocked=true;
            twoVerticalLockedX=null;
          }else if(travel>=10){
            if(absDY>=absDX){
              gestureAxis='vertical';
              twoVerticalLockedX=nearestTwoTargetX(camX);
            }else{
              gestureAxis='free';
              twoHorizontalUnlocked=true;
              twoVerticalLockedX=null;
            }
          }
        }
      }

      if(gestureAxis==='vertical'){
        const recentSide=
          absStepX>=3 &&
          absStepX>Math.max(absStepY*1.28,3);

        const accumulatedSide=
          absDX>=20 &&
          absDX>absDY*.76;

        if(recentSide || accumulatedSide) sideIntentFrames++;
        else sideIntentFrames=Math.max(0,sideIntentFrames-1);

        if(sideIntentFrames>=2){
          // User clearly changed intent to a side/free canvas drag.
          gestureAxis='free';
          twoHorizontalUnlocked=true;
          twoVerticalLockedX=null;

          // Re-anchor at the current visual position so there is no jump.
          dragStartX=e.clientX;
          dragStartY=e.clientY;
          dragStartCamX=camX;
          dragStartCamY=camY;
          dx=0;
          dy=0;
        }else{
          if(twoVerticalLockedX===null){
            twoVerticalLockedX=nearestTwoTargetX(camX);
          }

          // Vertical feed scrolling is fixed on the nearest complete
          // two-column page. No sideways drift.
          applyDragResistance(
            twoVerticalLockedX,
            dragStartCamY-dy
          );
        }
      }

      if(gestureAxis==='free'){
        // Same direct finger-following feel as the borderless canvas.
        applyDragResistance(
          dragStartCamX-dx,
          dragStartCamY-dy
        );
      }

      if(gestureAxis==='pending'){
        // Before intent is decided, allow vertical movement only.
        // This prevents micro horizontal jitter.
        applyDragResistance(
          dragStartCamX,
          dragStartCamY-dy
        );
      }

      const now=performance.now();
      const dt=Math.max(1,now-lastT);

      velX=gestureAxis==='free'
        ? -(e.clientX-lastX)/dt
        : 0;
      velY=-(e.clientY-lastY)/dt;

      lastX=e.clientX;
      lastY=e.clientY;
      lastT=now;

      render();
      e.preventDefault();
      return;
    }
  });

  function endPointer(e){
    const wasDragPointer=e.pointerId===dragId;
    const tapDistance=Math.hypot(e.clientX-tapStartX,e.clientY-tapStartY);
    const tapDuration=performance.now()-tapStartTime;
    const tappedCard=(
      wasDragPointer &&
      pointers.size===1 &&
      tapCandidateIndex>=0 &&
      !tapMoved &&
      tapDistance<=9 &&
      tapDuration<=650 &&
      !pinch &&
      !detailOpen
    ) ? tapCandidateIndex : -1;

    pointers.delete(e.pointerId);

    if(tappedCard>=0){
      dragId=null;
      surface.classList.remove('dragging');

      velX=velY=0;
      tapCandidateIndex=-1;
      tapMoved=false;

      cancelBorderlessCenter(true);
      stopMotion();
      openMinimalDetail(tappedCard);
      return;
    }

    tapCandidateIndex=-1;
    tapMoved=false;

    if(pinch && pointers.size<2){
      // Enough physical intent commits; otherwise smoothly returns.
      settlePinch(pinch.progress>=.30);
      dragId=null;
      return;
    }

    if(e.pointerId===dragId){
      dragId=null;
      surface.classList.remove('dragging');

      const ix=velX*16.67;
      const iy=velY*16.67;

      if(mode==='two'){
        // Never remember an old page before the glide finishes.
        // Recovery is calculated from the FINAL resting position.
        twoRecoveryTargetX=null;

        if(Math.hypot(ix,iy)>.40){
          inertia(ix,iy,false);
        }else{
          scheduleTwoAlignment(180,nearestTwoTargetX(camX));
        }
      }else if(Math.hypot(ix,iy)>.55){
        inertia(ix,iy,false);
      }else if(mode==='borderless'){
        scheduleBorderlessCenter(180);
      }else{
        springToBounds();
      }
    }

    if(pointers.size===1 && !pinch){
      const [id,p]=[...pointers.entries()][0];
      dragId=id;
      dragStartX=lastX=p.x;
      dragStartY=lastY=p.y;
      dragStartCamX=camX;
      dragStartCamY=camY;
      lastT=performance.now();
      velX=velY=0;
      gestureAxis='pending';
      sideIntentFrames=0;
      twoHorizontalUnlocked=false;
      twoVerticalLockedX=null;
      surface.classList.add('dragging');
    }
  }

  surface.addEventListener('pointerup',endPointer);
  surface.addEventListener('pointercancel',endPointer);

  /* =================================================================
     06 · TRACKPAD / WHEEL
     ================================================================= */
  let wheelAxis='pending';
  let wheelAccumX=0,wheelAccumY=0,wheelSideFrames=0;
  let wheelLockedX=null;
  let wheelTimer=0;

  function resetWheel(){
    wheelAxis='pending';
    wheelAccumX=wheelAccumY=0;
    wheelSideFrames=0;
    wheelLockedX=null;
  }

  surface.addEventListener('wheel',e=>{
    if(e.ctrlKey){
      // Desktop test: ctrl-wheel simulates two-hand scale.
      const r=surface.getBoundingClientRect();
      const anchor={x:e.clientX-r.left,y:e.clientY-r.top};
      const subject=nearestCard(anchor.x,anchor.y);
      borderlessSubject=subject;

      const from=mode;
      const to=e.deltaY<0?'borderless':'two';
      if(to!==from){
        const fromLayout=from==='two'?twoLayout():borderlessLayout(subject);
        const toLayout=to==='two'?twoLayout():borderlessLayout(subject);
        const subjectTo=toLayout[subject];

        layout=toLayout;
        mode=to;
        camX=subjectTo.x+subjectTo.w/2-anchor.x;
        camY=subjectTo.y+subjectTo.h/2-anchor.y;
        surface.classList.toggle('two',mode==='two');
        surface.classList.toggle('borderless',mode==='borderless');
        render();
      }
      e.preventDefault();
      return;
    }

    stopMotion();

    const dx=e.shiftKey && Math.abs(e.deltaX)<.01?e.deltaY:e.deltaX;
    const dy=e.shiftKey && Math.abs(e.deltaX)<.01?0:e.deltaY;

    if(mode==='borderless'){
      cancelBorderlessCenter(true);
      applyDragResistance(camX+dx,camY+dy);
      render();

      clearTimeout(wheelTimer);
      wheelTimer=setTimeout(()=>{
        scheduleBorderlessCenter(180);
        resetWheel();

    wheelAccumY+=Math.abs(dy);

    }

    if(wheelAxis==='vertical'){
      const side=Math.abs(dx)>=2.5 && Math.abs(dx)>Math.max(Math.abs(dy)*1.10,
