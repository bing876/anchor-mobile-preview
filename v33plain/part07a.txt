2.5);
      const accumulated=wheelAccumX>=18 && wheelAccumX>wheelAccumY*.34;

      if(side||accumulated)wheelSideFrames++;
      else wheelSideFrames=Math.max(0,wheelSideFrames-1);

      if(wheelSideFrames>=2)wheelAxis='free';
    }

    if(mode==='two'){
      if(wheelAxis==='vertical'){
        if(wheelLockedX===null){
          wheelLockedX=nearestTwoTargetX(camX);
        }
        applyDragResistance(wheelLockedX,camY+dy);
      }else{
        wheelLockedX=null;
        applyDragResistance(camX+dx,camY+dy);
      }

      render();

      clearTimeout(wheelTimer);
      wheelTimer=setTimeout(()=>{
        const finalTarget=nearestTwoTargetX(camX);
        twoRecoveryTargetX=null;
        scheduleTwoAlignment(220,finalTarget);
        resetWheel();
      },220);

      e.preventDefault();
      return;
    }

    if(wheelAxis==='vertical'){
      applyDragResistance(camX,camY+dy);
    }else{
      applyDragResistance(camX+dx,camY+dy);
    }

    render();

    clearTimeout(wheelTimer);
    wheelTimer=setTimeout(()=>{
      springToBounds();
      resetWheel();
    },120);

    e.preventDefault();
  },{passive:false});
